```c
F(int r[], int s, int t)
{
    for (int j=2*s; j<=t; j*=2)
    {
        if (r[j+1]<r[j]) ++j;
        if (r[s]<=r[j]) break;
        int x=r[s];
        r[s]=r[j];
        r[j]=x;
        s = j;
    }
}
```
